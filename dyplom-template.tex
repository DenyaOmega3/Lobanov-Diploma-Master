\chapter{Максимальне кубічне кільце}\label{chaper1}

\section{Попередні відомості про кубічні кільця}\label{section1.1}
Необхідно пригадати важливі означення та результати.
\begin{definition}
\textbf{Кубічним кільцем} $\Lambda$ називають вільний $\mathbb{Z}$-модулем рангу $3$.
\end{definition}

Оскільки кубічне кільце містить елемент $1$, то буде зручніше надалі обирати базис вигляду $\{1,\omega_1,\omega_2\}$.

\begin{definition}
Базис $\{1,\omega_1,\omega_2\}$ кубічного кільця $\Lambda$ називається \textbf{нормальним}, якщо
\begin{center}
$\omega_1 \cdot \omega_2 \in \mathbb{Z}$.
\end{center}
\end{definition}

\begin{remark}
Зафіксуємо базис $\{1,\omega_1,\omega_2\}$ кільця $\Lambda$. Тоді $\omega_1 \cdot \omega_2 = k \omega_1 + l \omega_2 + m$ при $k,l,m \in \mathbb{Z}$. Замінивши $\omega_1$ на $\omega_1-l$ та $\omega_2$ на $\omega_2 - k$, отримуємо нормальний базис $\{1,\omega_1 - l, \omega_2 - k\}$.
\end{remark}

Отже, \textcolor{red}{надалі всі базиси в кубічних кільцях -- нормальні}.

Запишемо таблицю множення для базиса $\{1,\omega_1,\omega_2\}$:
\begin{align*}
\omega_1^2 & = b \omega_1 + a \omega_2 + r \\
\omega_1 \omega_2 & = s \\
\omega_2^2 & = d \omega_1 + c \omega_2 + t
\end{align*}
Прирівнявши $(\omega_1^2) \cdot \omega_2 = \omega_1 \cdot (\omega_1 \cdot \omega_2)$ та $\omega_1 \cdot (\omega_2^2) = (\omega_1 \cdot \omega_2) \cdot \omega$, отримуємо співвідношення $r = -ac,\ s = ad,\ t = -bd$. Отже, таблиця множення задається чотирма числами $a,b,c,z \in \mathbb{Z}$:
\begin{align*}
\omega_1^2 & = b \omega_1 + a \omega_2 - ac \\
\omega_1 \omega_2 & = ad \\
\omega_2^2 & = d \omega_1 + c \omega_2 + -bd
\end{align*}

\begin{definition}
Нехай $\Lambda$ -- кубічне кільце та $M \subset \Lambda$ -- решітка, тобто $M = \left<\theta_1,\theta_2,\theta_3\right>$. Розкладемо кожний $\theta_i$ за базисом $\{1,\omega_1,\omega_2\}$ таким чином:
\begin{align*}
\theta_1 = c_{10} \cdot 1 + c_{11} \cdot \omega_1 + c_{12} \omega_2 \\
\theta_2 = c_{20} \cdot 1 + c_{21} \cdot \omega_1 + c_{22} \omega_2 \\
\theta_3 = c_{30} \cdot 1 + c_{31} \cdot \omega_1 + c_{32} \omega_2
\end{align*}
\textbf{Індексом решітки $M$} називатимемо значення:
\begin{align*}
[\Lambda : M] = \left| \det \begin{pmatrix}
c_{10} & c_{20} & c_{30} \\
c_{11} & c_{21} & c_{31} \\
c_{12} & c_{22} & c_{32} \\
\end{pmatrix} \right|.
\end{align*}
\end{definition}

Розглянемо кубічне кільце $\Lambda = \left<1,\omega_1,\omega_2\right>$, а також решітку $\mathbb{Z}(\alpha) = \left<1,\alpha,\alpha^2\right>$, де елемент $\alpha = z + x \omega_1 + y \omega_2$. Знайдемо індекс $[\Lambda : \mathbb{Z}(\alpha)]$. Детально розпишемо елемент $\alpha^2$.
\begin{align*}
\alpha^2 & = (z + x \omega_1 + y \omega_2)^2 = z^2 + x^2 \omega_1^2 + y^2 \omega_2^2 + 2zx \omega_1 + 2zy \omega_2 + 2 xy \omega_1 \omega_2 \\
& = z^2 + x^2 (b \omega_1 + a \omega_2 - ac) +y^2 (d \omega_1 + c \omega_2 - bd) \\ 
& + 2zx \omega_1 + 2yz \omega_2 + 2xyad
\end{align*} 
Отже, маємо ось такий розклад:
\begin{multline*}
1 = 1 \cdot 1 + 0 \cdot \omega_1 + 0 \cdot \omega_2 \\
\alpha = z \cdot 1 + x \omega_1 + y \omega_2 \\
\alpha^2 = (z^2 - x^2 ac - y^2 bd + 2xyad) \cdot 1 + (x^2 b + y^2 d + 2xz) \omega_1 + (x^2 a + y^2 c + 2yz) \omega_2
\end{multline*}
Отже, індекс обчислюється таким чином:
\begin{align*}
[\Lambda : \mathbb{Z}(\alpha)] & = \left| \det \begin{pmatrix}
1 & z & z^2-x^2ac-y^2bd+2xyad \\
0 & x & x^2b+y^2d+2xz \\
0 & y & x^2a+y^2c+2yz
\end{pmatrix} \right| \\
& = |x(x^2a+y^2c+2yz) -y(x^2b+y^2d+2xz)| \\
& = |ax^3 + cxy^2 - bx^2y - dy^3|.
\end{align*}

\begin{definition}
\textbf{Кубічною формою} (або \textbf{індекформою}) називатимемо вираз
\begin{align*}
f(x,y) = ax^2 + cxy^2 - bx^2y - dy^2
\end{align*}
Його дискримінант обчислюється за формулою:
\begin{align*}
\Delta = b^2 c^2 - 4c^3 a - 4b^3 d + 18 abcd - 27a^2d^2.
\end{align*}
\end{definition}

У результаті отримали, що кожному кубічному кільцю $\Lambda$ відповідатиме з точністю до знаку кубічна форма $f(x,y)$.

І навпаки, будь-якою кубічною формою $f(x,y)$, що задається числами $a,b,c,d \in \mathbb{Z}$ з ненулевим дискримінантом, можна визначити кубічне кільце $\Lambda$.

Нехай $\Lambda$ -- кубічне кільце та $\alpha \in R$. Розглянемо оператор множення на елемент $\alpha$, тобто $x \mapsto \alpha x$. Якщо $\{\alpha_1,\alpha_2,\alpha_3\}$ -- базис кільця $\Lambda$ (довільний базис), то оператору відповідатиме матриця $[\alpha]$, від якої можна знайти слід $\tr \alpha$.
 
\begin{definition}
\textbf{Дискримінантом кубічного кільця}  $\Lambda$ назвемо матрицю:
\begin{align*}
\Delta_\Lambda = \det \begin{pmatrix}
\tr(\alpha_1 \alpha_1) & \tr(\alpha_1 \alpha_2) & \tr(\alpha_1 \alpha_3) & \\
\tr(\alpha_2 \alpha_1) & \tr(\alpha_2 \alpha_2) & \tr(\alpha_2 \alpha_3) & \\
\tr(\alpha_3 \alpha_1) & \tr(\alpha_3 \alpha_2) & \tr(\alpha_3 \alpha_3) & \\
\end{pmatrix}
\end{align*}
\end{definition}

\begin{remark}
Дискримінант кільця $\Delta_\Lambda$ не залежить від вибору базису. Тому надалі можемо працювати з нормальним базисом $\{1,\omega_1,\omega_2\}$. Маючи даний базис та таблицю множення, отримуємо
\begin{align*}
\Delta_\Lambda & = \det \begin{pmatrix}
\tr 1 & \tr \omega_1 & \tr \omega_2 \\
\tr \omega_1 & \tr \omega_1^2 & \tr \omega_1 \omega_2 \\
\tr \omega_2 & \tr \omega_1 \omega_2 & \tr \omega_2^2
\end{pmatrix} = \det \begin{pmatrix}
3 & b & c \\
b & b^2-2ac & 3ad \\
c & 3ad & c^2-2bd
\end{pmatrix} \\
& = b^2 c^2 - 4c^3 a - 4b^3 d + 18abcd - 27a^2d^2 = \Delta.
\end{align*}
Отже, дискримінант кубічного кільця $\Lambda$ дорівнює дискримінанту кубічної форми $f(x,y)$.
\end{remark}

\section{Критерій максимальності кільця}
\begin{definition}
Кубічне кільце $\Lambda$ будемо називати \textbf{немаксимальним}, якщо існує більше кубічне кільце $\tilde{\Lambda} \supsetneq \Lambda$. Іншими словами, існує елемент $\theta \in \tilde{\Lambda}$, при цьому $\theta \notin \Lambda$.
\end{definition}

Якщо кубічне кільце $\Lambda$ є немаксимальним, то, взявши елемент $\theta \notin \Lambda$, можемо розглянути розширення $\tilde{\Lambda} = \Lambda[\theta]$. З'ясуємо, за яких умов $\tilde{\Lambda}$ стане кубічним кільцем також.

\subsection{$\theta^3 \in \tilde{\Lambda}$}
Елемент $\rho \in \tilde{\Lambda}$ записується як многочлен із $\mathbb{Z}[\theta]$.  Оскільки нас цікавить модуль рангу $3$, один із варіантів -- це вимагати, щоб $\theta^3 \in \tilde{\Lambda}$. У такому разі кожний елемент записується в такому вигляді:
\begin{align*}
\rho = x + y \theta + z \theta^2.
\end{align*}
Зрозуміло, що $\{ 1,\theta,\theta^2 \}$ утворює базис кільця $\tilde{\Lambda}$. Оскільки $\theta^3 \in \tilde{\Lambda}$ за припущенням, то даний базис задає кубічне кільце $\tilde{\Lambda}$.

Оскільки $\omega_1,\omega_2 \in \Lambda$, то звідси $\omega_1,\omega \in \tilde{\Lambda}$ та записуються в лінійну комбінацію $1,\theta,\theta^2$.  У такому разі елемент $\theta$ зобов'язаний мати вигляд:
\begin{align*}
\theta = \dfrac{\alpha + \beta \omega_1 + \gamma \omega_2}{p}
\end{align*}
при $\alpha,\beta,\gamma \in \mathbb{Z}$ та також $p \in \mathbb{Z}$. 

\begin{remark}
Вибір $\alpha,\beta,\gamma$ може бути обмеженим до $\pmod p$.
\end{remark}

Справді, нехай $\Lambda = \left< 1,\omega_1,\omega_2\right>$ та припустимо ми маємо два розширення $\tilde{\Lambda}_1 = \left< 1, \theta, \theta^2\right>$ та $\tilde{\Lambda}_2 = \left< 1, \sigma, \sigma^2\right>$, де елементи $\theta = \dfrac{\beta \omega_1 + \gamma \omega_2 + \alpha}{p}$ та $\sigma = \dfrac{\beta' \omega_1 + \gamma' \omega_2 + \alpha'}{p}$, причому $\alpha' \equiv \alpha,\ \beta' \equiv \beta,\ \gamma' \equiv \gamma \pmod p$. Іншими словами, $\alpha' = \alpha + px,\ \beta' = \beta + py,\ \gamma' = \gamma + pz$ при деяких $x,y,z \in \mathbb{Z}$. Розпишемо $\sigma$ на основі цих рівностей:
\begin{align*}
\sigma & = \dfrac{\beta \omega_1 + py \omega_1 + \gamma \omega_2 + pz \omega_2 + \alpha + px}{p} \\
& = \dfrac{\beta \omega_1 + \gamma \omega_2 + \alpha}{p} + y \omega_1 + z \omega_2 + x = \theta + x + y \omega_1 + z \omega_2.
\end{align*}
Оскільки $1,\omega_1,\omega_2 \in \tilde{\Lambda}_1$ та $\theta \in \tilde{\Lambda}_1$, то звідси $\sigma \in \tilde{\Lambda}_1$. Водночас окільки $1,\omega_1,\omega_2 \in \tilde{\Lambda}_2$ та $\sigma \in \tilde{\Lambda}_2$, то, виразивши $\theta$, отримуємо $\theta \in \tilde{\Lambda}_2$.

Отже, отримуємо $\tilde{\Lambda}_1 = \tilde{\Lambda}_2$.

\textcolor{red}{Надалі обмежуємось випадком, коли $\gamma = 0$}.

Час відповісти на питання, коли $\theta^3 \in \tilde{\Lambda}$. Для цього розглянемо оператор множення на елемент $\theta$, тобто $\lambda \mapsto \theta \cdot \lambda$. З'ясуємо, як буде виглядати матриця $[\theta]$ даного оператора.
\begin{align*}
\theta \cdot 1 = \dfrac{\alpha + \beta \omega_1}{p} = \dfrac{\alpha}{p} \cdot 1 + \dfrac{\beta}{p} \omega_1 + 0 \cdot \omega_2
\end{align*}
\begin{align*}
\theta \cdot \omega_1 & = \dfrac{\alpha + \beta \omega_1}{p} \cdot \omega_1 = \dfrac{\alpha \omega_1}{p} + \dfrac{\beta (-ac + b \omega_1 + a \omega_2)}{p} = \\
& = \dfrac{-\beta a c}{p} \cdot 1 + \dfrac{\alpha + \beta b}{p} \cdot \omega_1 + \dfrac{\beta a}{p} \cdot \omega_2
\end{align*}
\begin{align*}
\theta \cdot \omega_2 = \dfrac{\alpha + \beta \omega_1}{p} \cdot \omega_2 = \dfrac{\alpha \omega_2 + \beta ad}{p} = \dfrac{\beta a d}{p} \cdot 1 + 0 \cdot \omega_1 + \dfrac{\alpha}{p} \cdot \omega_2
\end{align*}

Зібравши всі коефіцієнти, отримуємо такий вигляд матриці:
\begin{align*}
[\theta] = \dfrac{1}{p} \begin{pmatrix}
\alpha & -\beta a c & \beta a d \\
\beta & \alpha + \beta b & 0 \\
0 & \beta a & \alpha
\end{pmatrix}
\end{align*}

Запишемо характеристичний многочлен даної матриці:
\begin{align*}
\chi_{[\theta]}(t) = \det [\theta] - M_{2 \times 2}[\theta] \cdot t + \tr [\theta] \cdot t^2 - t^3,
\end{align*}
де
\begin{align*}
M_{2 \times 2}[\theta] = M_{12}^{12}[\theta] + M_{13}^{13}[\theta] + M_{23}^{23}[\theta].
\end{align*}
Іншими словами, $M_{2 \times 2}[\theta]$ -- сума всіх діагональних мінорів $2 \times 2$.

Отже, $\chi_{[\theta]}(\theta) = 0$. Це означає, $\theta^3 = \det [\theta] - M_{2 \times 2}[\theta] \cdot \theta + \tr [\theta] \cdot \theta^2$. Оскільки нам необхідно $\theta^3 \in \left<1,\theta,\theta^2\right>$, треба вимагати, щоб
\begin{align*}
\det [\theta],\ M_{2 \times 2}[\theta],\ \tr [\theta] \in \mathbb{Z}.
\end{align*}

Обчислимо кожний коефіцієнт.

\begin{align*}
\tr [\theta] = \dfrac{1}{p} (3 \alpha + \beta b) \\
\det [\theta] = \dfrac{1}{p^3} \left( \alpha^3 + \alpha^2 \beta b + \beta^3 a^2 d + \alpha \beta^2 ac \right)
\end{align*}
Детальніше розпишемо знаходження $M_{2 \times 2}[\theta]$. Маємо
\begin{align*}
M_{2 \times 2}[\theta] & = M_{12}^{12}[\theta] + M_{13}^{13}[\theta] + M_{23}^{23}[\theta] = \\
& = \dfrac{1}{p^2} \det \begin{pmatrix}
\alpha & -\beta \alpha c \\
\beta & \alpha + \beta b
\end{pmatrix} + \dfrac{1}{p^2} \det \begin{pmatrix}
\alpha & \beta a \alpha \\
0 & \alpha
\end{pmatrix} + \dfrac{1}{p^2} \det \begin{pmatrix}
\alpha + \beta b & 0 \\
\beta a & \alpha
\end{pmatrix} = \\
& = \dfrac{1}{p^2} \left( \alpha^2 + \alpha \beta b + \beta^2 \alpha c + \alpha^2 + \alpha^2 + \alpha \beta b \right) = \\ 
& = \dfrac{1}{p^2} \left( 3 \alpha^2 + \beta^2 a c + 2 \alpha \beta b\right)
\end{align*}

Із урахуванням необіхдних умов на ці числа, запишемо цю умову в термінах конгруентних рівнянь:
\begin{equation}
    \begin{cases}
      \alpha^3 + \alpha^2 \beta b + \beta^3 a^2 d + \alpha \beta^2 a c & \equiv 0 \pmod {p^3} \\
    3\alpha^2 + \beta^2 ac + 2 \alpha \beta b & \equiv 0 \pmod {p^2} \\
      3 \alpha + \beta b & \equiv 0 \pmod p 
    \end{cases}\,.
\end{equation}
\textcolor{red}{Надалі припускаємо, що $p$ -- просте число та $\alpha \neq 0$.} Тоді можна підібрати елемент $\alpha^{-1}$ -- оборотний елемент класу лишків $\mathbb{Z}_p$ відносно множення. Третє, друге, перше рівняння домножуємо на відповідно $\alpha^{-1},\ \alpha^{-2},\ \alpha^{-3}$ -- отримуємо наступне:
\begin{equation*}
    \begin{cases}
      1 + (\beta \alpha^{-1}) b + (\beta \alpha^{-1})^{3} a^2 d + (\beta \alpha^{-1})^2 a c & \equiv 0 \pmod {p^3} \\
    3 + ((\beta \alpha^{-1})^2 ac + 2 (\beta \alpha^{-1}) b & \equiv 0 \pmod {p^2} \\
      3 + (\beta \alpha^{-1})b & \equiv 0 \pmod p 
    \end{cases}\,.
\end{equation*}
Проведемо заміну:
\begin{align}
t = \beta \alpha^{-1}
\end{align}
Задача звелась до розв'язання системи конгруентних рівнянь відносно $t$, якщо $p$ -- просте число: 
\begin{equation}
\label{system_congruent_equations}
    \begin{cases}
      a^2 d t^3 + ac t^2 + bt + 1 & \equiv 0 \pmod {p^3} \\
      act^2 + 2bt + 3 & \equiv 0 \pmod {p^2} \\
      3 + bt & \equiv 0 \pmod p 
    \end{cases}\,.
\end{equation}

\iffalse
Підіб'ємо підсумок:
\begin{proposition}
Нехай $\Lambda$ -- кубічне кільце, яке задається числами $(a,b,c,d)$. Тоді $\Lambda$ -- немаксимальне $\iff$ існує розв'язок системи рівнянь \eqref{system_congruent_equations}.
\end{proposition}
\fi
На цьому ще не все. Розпишемо елемент $\omega_1$ через лінійну комбінацію $\{1,\theta,\theta^2\}$. Маємо наступне:
\begin{align*}
\omega_1 = \xi \cdot 1 + \eta \cdot \dfrac{\alpha + \beta \omega_1}{p} + \zeta \cdot \dfrac{\alpha^2 + 2 \alpha \beta \omega_1 + \beta^2 (b \omega_1 + a \omega_2 - ac)}{p^2}
\end{align*}
Прирівнюючи коефіцієнти, отримуємо наступне:
\begin{align*}
\xi + \eta \cdot \dfrac{\alpha}{p} + \zeta \cdot \dfrac{\alpha^2 - \beta^2 ac}{p^2} = 0 \\
\eta \cdot \dfrac{\beta}{p} + \zeta \cdot \dfrac{2\alpha\beta + \beta^2 b}{p^2} = 1 \\
\zeta \cdot \dfrac{\beta^2 a}{p^2} = 0
\end{align*}
Отже, звідси $\zeta = 0$, а також $\eta = \dfrac{p}{\beta}$. Проте оскільки $p$ -- просте число за припущенням, то $\beta \in \{1,p\}$. Далі, маємо $\xi = - \dfrac{\alpha}{\beta}$.

Якщо $\beta = 1$, то звідси $\xi = -\alpha$, отримуємо цілий коефіцієнт.

Якщо $\beta = p$, то звідси $\xi = -\dfrac{\alpha}{p}$. Отже, вимагаємо $\alpha \equiv 0 \pmod p$. Проте в цьому випадку $\theta = \dfrac{\alpha + \beta \omega_1}{p} = \dfrac{\alpha}{p} + \omega_1 \in \Lambda$. Тобто буде $\Lambda = \tilde{\Lambda}$, не виникне розширення.

\begin{lemma}
Нехай $\Lambda$ -- кубічне кільце та $\theta = \dfrac{\alpha + \beta \omega_1}{p}$ при $\beta \neq 1$. Тоді кільце не допускає розширення $\tilde{\Lambda} = \left< 1,\theta,\theta^2\right>$.
\end{lemma}

\textcolor{red}{Надалі припускаємо, що $\beta = 1$.}

Аналогічно проведемо процедуру для $\omega_2$. Отримуємо:
\begin{align*}
\xi + \eta \cdot \dfrac{\alpha}{p} + \zeta \cdot \dfrac{\alpha^2 - \beta^2 ac}{p^2} = 0 \\
\eta \cdot \dfrac{\beta}{p} + \zeta \cdot \dfrac{2\alpha\beta + \beta^2 b}{p^2} = 0 \\
\zeta \cdot \dfrac{\beta^2 a}{p^2} = 1
\end{align*}

Звідси $\zeta = \dfrac{p^2}{a}$. Отже, $a \in \{1,p,p^2\}$. Далі, $\eta \cdot \dfrac{1}{p} + \dfrac{p^2}{a} \cdot \dfrac{2\alpha + b}{p^2} = 0$. Звідси $\eta = -\dfrac{p(2\alpha+b)}{a}$. Нарешті, $\xi = \dfrac{\alpha^2 + \alpha b + ac}{a}$.

\begin{lemma}
Нехай $\Lambda$ -- кубічне кільце та $\theta = \dfrac{\alpha + \omega_1}{p}$. Тоді кільце не допускає розширення $\tilde{\Lambda} = \left< 1,\theta,\theta^2\right>$, якщо $a \notin \{1,p,p^2\}$.
\end{lemma}

Розглянемо кожний $a$ окремо.
\subsubsection{$a = 1$} Усі числа $\zeta,\eta,\xi$ -- цілі, тож не потребує додаткових обмежень.

\subsubsection{$a = p$} Тоді $\zeta,\eta$ -- цілі, але $\xi = \dfrac{\alpha^2 + \alpha b + ac}{p}$, тож додатково треба додати обмеження $\alpha^2 + \alpha b + ac \equiv 0 \pmod p$. Маючи умову $a = p$, ми отримуємо
\begin{align*}
\alpha(\alpha+b) \equiv 0 \pmod p \iff \left[ \begin{gathered} \alpha \equiv 0 \pmod p \\ \alpha \equiv -b \pmod p \end{gathered} \right. .
\end{align*}
Якщо $\alpha \equiv 0 \pmod p$, то із системи рівнянь випливає, що $p \mid b,c,d$.

\begin{lemma}
Нехай $\Lambda$ -- кубічне кільце та $a = p$, а також $p \mid b,c,d$. Тоді кільце $\tilde{\Lambda} = \left<1,\theta,\theta^2\right>$ буде розширенням при $\theta = \dfrac{\omega_1}{p}$.
\end{lemma}

Якщо $\alpha \equiv -b \pmod p$, то із третього рівняння системи отримуємо $-2b \equiv 0 \pmod p$. Якщо $p \neq 2$, то тоді $p \mid b$, але тоді знову $\alpha \equiv 0 \pmod p$. Отримуємо такі самі обмеження.

Утім при $\alpha \equiv -b \pmod p$ та при $p = 2$ ситуація змінюється. Залишається варіант -- коли $\alpha = 1$, а тому $b \equiv 1 \pmod 2$. Утім із другого рівняння системи випливає, що $3 + 2c + 2b \equiv 0 \pmod 4$, проте якщо спуститися до $\pmod 2$, ми отримуємо $1 \equiv 0 \pmod 2$, що неможливо. 

\subsubsection{$a = p^2$} Тоді $\zeta$ -- ціле, решта $\eta = -\dfrac{2\alpha + b}{p}$ та $\xi = \dfrac{\alpha^2 + \alpha b + ac}{p^2}$. Отже, маємо два додаткових обмеження:
\begin{align*}
-2\alpha - b \equiv 0 \pmod p \\
\alpha^2 + \alpha b + ac \equiv 0 \pmod {p^2}
\end{align*}
Отримуємо $\alpha \equiv 0 \pmod p$. Внаслідок чого $b \equiv 0 \pmod p$. Всі решта рівняння виконуються.

\begin{lemma}
Нехай $\Lambda$ -- кубічне кільце та $a = p^2$, а також $p \mid b$. Тоді кільце $\tilde{\Lambda} = \left<1,\theta,\theta^2\right>$ буде розширенням при $\theta = \dfrac{\omega_1}{p}$.
\end{lemma}

Нормалізуємо новоутворене кубічне кільце $\tilde{\Lambda}$. Маємо наступне:
\begin{align*}
\theta \cdot \theta^2 = \theta^3 = \det [\theta] - M_{2 \times 2}[\theta] \cdot \theta + \tr[\theta] \cdot \theta^2
\end{align*}
Оберемо новий базис $\left\{ 1,\ \theta-\tr[\theta],\ \theta^2 + M_{2 \times 2}[\theta] \right\}$. Для зручності позначу цей базис за $\{1,\tilde{\omega}_1,\tilde{\omega}_2\}$. У результаті за процедурою нормалізації базису вище отримуємо
\begin{align*}
\tilde{\omega}_1 \cdot \tilde{\omega}_2 = \det [\theta] - \tr [\theta] M_{2 \times 2}[\theta]
\end{align*}
Визначимо, які числа $\tilde{a},\tilde{b},\tilde{c},\tilde{d}$ визначені в кільці $\tilde{\Lambda} = \left< 1, \tilde{\omega}_1,\tilde{\omega}_2 \right>$.
\begin{align*}
\tilde{\omega}_1^2 = (\theta - \tr [\theta])^2 = \theta^2 - 2 \theta \tr [\theta] + (\tr [\theta])^2 \\ = \tilde{b} (\theta - \tr [\theta]) + \tilde{a} (\theta^2 + M_{2 \times 2}[\theta]) - \tilde{a} \tilde{c} = \tilde{b} \tilde{\omega}_1 + \tilde{a} \tilde{\omega}_2 - \tilde{a} \tilde{c}
\end{align*}
Звідси випливає, що $\tilde{a} = 1,\ \tilde{b} = - 2 \tr [\theta]$ та $\tilde{c} = \tr^2 [\theta] + M_{2 \times 2}[\theta]$. Далі,
\begin{align*}
\tilde{\omega}_1 \cdot \tilde{\omega}_2 = (\theta - \tr [\theta])(\theta^2 + M_{2 \times 2}[\theta]) =\\ = (\det [\theta] - M_{2 \times 2}[\theta] \theta + \tr [\theta] \theta^2) + M_{2 \times 2}[\theta] \theta - \tr[\theta] \theta^2 - \tr [\theta] M_{2 \times 2}[\theta] \\
= \det [\theta] - \tr [\theta] M_{2 \times 2}[\theta] = \tilde{a} \tilde{d}
\end{align*}
Звідси $\tilde{d} = \det [\theta] - \tr [\theta] M_{2 \times 2}[\theta]$.

Отже, кубічне кільце $\tilde{\Lambda}$ задається числами
\begin{align*}
(\tilde{a},\tilde{b},\tilde{c},\tilde{d}) = (1,\ -2 \tr [\theta],\ \tr^2[\theta] + M_{2 \times 2}[\theta],\ \det [\theta] - \tr[\theta] M_{2 \times 2}[\theta]).
\end{align*}

\subsection{$\theta^3 \notin \tilde{\Lambda}$}
Увесь цей час, маючи кільце $\Lambda$, ми розширювали кільце $\tilde{\Lambda}$ таким чином, що $\{1,\theta,\theta^2\}$ був обов'язково базисом. Чи можна розширити кільце так, щоб в базисі лише один елемент не потрапляв у $\tilde{\Lambda}$?

Розглянемо $\theta \notin \Lambda$. Елементи $\tilde{\Lambda}$ матимуть вигляд $x + y \omega_1 + z \theta$, тобто припускаємо, що $\{1,\omega_1,\theta\}$ буде базисом. Тоді оскільки $\omega_2 \in \Lambda$, то елемент $\theta$ можна подати в вигляді
\begin{align*}
\theta = \dfrac{\alpha + \beta \omega_1 + \omega_2}{p}
\end{align*}
Для кубічного кільця треба, щоб $\omega_1 \theta$ та $\theta^2$ потрапляли в $\tilde{\Lambda}$.
\begin{align*}
\omega_1 \theta & = \omega_1 \dfrac{\alpha + \beta \omega_1 + \omega_2}{p} = \dfrac{ad + \alpha \omega_1 + \beta (b\omega_1 + a\omega_2 - ac)}{p} \\
& = x + y \omega_1 + z \dfrac{\alpha + \beta \omega_1 + \omega_2}{p}
\end{align*}
Шукаючи числа $x,y,z$, отримуємо наступне:
\begin{align*}
x = \dfrac{ad - \beta ac - \alpha \beta a}{p} \\
y = \dfrac{\alpha + \beta b - \beta^2 a}{p} \\
z = \beta a
\end{align*}
Оскільки $x,y \in \mathbb{Z}$, то звідси маємо
\begin{equation*}
\begin{cases}
ad - \beta ac - \alpha \beta a & \equiv 0 \pmod p \\
\alpha + \beta b - \beta^2 a & \equiv 0 \pmod p
\end{cases}
\end{equation*}
Проведемо схожу процедуру з $\theta^2$.
\begin{align*}
\theta^2 & = \dfrac{\alpha^2 + \beta^2 (b \omega_1 + a \omega_2 - ac) + (d \omega_1 + c \omega_2 - bd) + 2 \alpha \beta \omega_1 + 2 \alpha \omega_2 + 2 \beta ad }{p^2} \\
& = x + y \omega_1 + z \dfrac{\alpha + \beta \omega_1 + \omega_2}{p}
\end{align*}
Шукаючи коефіцієнти, отримуємо
\begin{align*}
x = \dfrac{-\alpha^2 - \beta^2 ac - bd + 2 \beta a d - \alpha c - \alpha \beta^2 a}{p^2} \\
y = \dfrac{\beta^2 b + d - \beta^2 c - \beta^3 a}{p^2} \\
z = \dfrac{c +\beta^2 a + 2\alpha}{p}
\end{align*}
Оскільки знову $x,y,z \in \mathbb{Z}$, маємо отримати
\begin{equation*}
\begin{cases}
c +\beta^2 a + 2\alpha & \equiv 0 \pmod p \\
\beta^2 b + d - \beta^2 c - \beta^3 a & \equiv 0 \pmod {p^2} \\
-\alpha^2 - \beta^2 ac - bd + 2 \beta a d - \alpha c - \alpha \beta^2 a & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}
Разом маємо систему конгруентних рівнянь:
\begin{equation*}
\begin{cases}
ad - \beta ac - \alpha \beta a & \equiv 0 \pmod p \\
\alpha + \beta b - \beta^2 a & \equiv 0 \pmod p \\
c +\beta^2 a + 2\alpha & \equiv 0 \pmod p \\
\beta^2 b + d - \beta^2 c - \beta^3 a & \equiv 0 \pmod {p^2} \\
-\alpha^2 - \beta^2 ac - bd + 2 \beta a d - \alpha c - \alpha \beta^2 a & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}

\begin{remark}
Аналогічно нам досить розглядати лише $\alpha,\beta$ з точністю до $\pmod p$.
\end{remark}

\section{Розв'язання системи для $\theta^3 \notin \tilde{\Lambda}$.}
Припустимо, що $a \equiv 0 \pmod p$. Тоді система спроститься до вигляду:
\begin{equation*}
\begin{cases}
\alpha + \beta b & \equiv 0 \pmod p \\
c + 2\alpha & \equiv 0 \pmod p \\
\beta^2 b + d - \beta^2 c - \beta^3 a & \equiv 0 \pmod {p^2} \\
-\alpha^2 - \beta^2 ac - bd + 2 \beta a d - \alpha c - \alpha \beta^2 a & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}

\subsection*{$p = 2$} Друге рівняння дає $c \equiv 0 \pmod 2$, а третє рівняння дає $\alpha \equiv - \beta b \pmod 2$. Піднімемо їх за $\pmod 4$ -- тоді
\begin{align*}
c \equiv 2k \pmod 4 \\
\alpha \equiv - b\beta + 2m \pmod 4
\end{align*}
Підставивши в третє та четверте рівняння системи, отримуємо наступне:
\begin{align*}
\beta^2 b + d - 2 \beta^2 k - \beta^3 \alpha \equiv 0 \pmod 4 \\
b (a \beta^3 + b \beta^2 - d + 2\beta k) \equiv 0 \pmod 4
\end{align*}
Виразивши $a \beta^3 - d$ із першого рівняння та підставивиш до другого рівняння, прийдемо до
\begin{align*}
2b \beta (\beta b - \beta k + k) \equiv 0 \pmod 4
\end{align*}
Якщо $\beta \equiv 0 \pmod 2$, то дане рівняння виконується. Крім того, ми би отримали $\alpha \equiv 0 \pmod 2$ та $d \equiv 0 \pmod {4}$.

\begin{lemma}
Кубічне кільце $\Lambda$ має розширення $\tilde{\Lambda}$ елементом $\theta = \dfrac{\omega_2}{p} \iff p \mid a,\ p \mid c,\ p^2 \mid d $.\\
(при $p = 2$).
\end{lemma}

Тепер нехай $\beta \equiv 1 \pmod 2$. Звідси $\alpha \equiv b \pmod 2$. Крім того, $2b^2 \equiv 0 \pmod 4$. Звідси випливає, що $b \equiv 0 \pmod 2$. Отже, $\alpha \equiv 0 \pmod 2$. Останні два рівняння системи дають обмеження:
\begin{align*}
-a + b - c + d \equiv 0 \pmod 4 \\
bd \equiv 0 \pmod 4
\end{align*}

\begin{lemma}
Кубічне кільце $\Lambda$ має розширення елементом $\theta = \dfrac{\omega_1 + \omega_2}{p} \iff p \mid a,\ p \mid c,\ p \mid b,\ p^2 \mid bd,\ p^2 \mid -a+b-c+d$.\\
(при $p=2$)
\end{lemma}

\subsection*{$p \neq 2$}
Зауважимо, що $2\alpha \equiv c \pmod p$. Якщо $p \neq 2$, то дане рівняння має розв'язок відносно $\alpha$. Звідси випливає, що $2 \alpha + 2 \beta b \equiv 0 \pmod p$, тобто $c + 2 \beta b \equiv 0 \pmod p$. Отже, $2 \beta b \equiv -c \pmod p$. Останнє має розв'язок відносно $\beta$ лише тоді, коли $\gcd(2b,p) \mid c$.

Припустимо, що $b \equiv 0 \pmod p$. Тоді $c \equiv 0 \pmod p$. Внаслідок чого $\alpha \equiv 0 \pmod p$. Якщо третє рівняння системи спустити до $\pmod p$, то ми отримуємо $d \equiv 0 \pmod p$. У результаті залишається розв'язати таке рівняння:
\begin{align*}
\beta^2 b + d - \beta^2 c - \beta^3 a \equiv 0 \pmod {p^2}
\end{align*}
Оскільки відомо, що $p \mid a,b,c,d$, то можна переписати таким чином:
\begin{align*}
-\beta^3 \tilde{a} + \beta^2 \tilde{b} - \beta^2 \tilde{c} + \tilde{d} \equiv 0 \pmod p
\end{align*}
Якщо $\tilde{a} \equiv 0 \pmod p$, отримуємо спрощену версію рівняння:
\begin{align*}
(\tilde{b} - \tilde{c})\beta^2 + \tilde{d} \equiv 0 \pmod p.
\end{align*}
Чи існує розв'язок рівняння, залежить від символа Лежандра $\left( \dfrac{\tilde{d}}{p} \right)$.

Нехай тепер $\tilde{a} \not\equiv 0 \pmod p$. Маємо
\begin{align*}
-\beta^3 + (\tilde{b} -\tilde{c}) a^{-1} \beta^2  + \tilde{d} \tilde{a}^{-1} \equiv 0 \pmod p
\end{align*}
Задача полягає в розв'язанні кубічного рівняння. Його можна звести до кубічного рівняння з лінійним доданком, якщо провести заміну $\beta = y + \gamma$. Такий $\gamma$ існуватиме лише при $p \neq 3$. Далі залежить від символа Лежандра $\left( \dfrac{\Delta}{p} \right)$, де $\Delta$ -- дискримінант рівняння $y^3 + uy + v = 0$.

Якщо $p = 3$, то виникне $\tilde{b} \equiv \tilde{c} \pmod 3$, внаслідок чого
\begin{align*}
\beta^3 \equiv (\tilde{d} \tilde{a}^{-1}) \pmod 3
\end{align*}
Дане рівняння має розв'язок $\iff (\tilde{d} \tilde{a}^{-1})^{2} \equiv 1 \pmod 3$. Іншими словами, $\tilde{d}^2 \equiv \tilde{a}^2 \pmod 3$.
\bigskip \\

\textcolor{red}{Надалі $a \not\equiv 0 \pmod p$}. Нехай кільце $\Lambda$ задовольняє умову $p \nmid c$. Маємо систему
\begin{equation*}
\begin{cases}
d - \beta c - \alpha \beta & \equiv 0 \pmod p \\
\alpha + \beta b - \beta^2 a & \equiv 0 \pmod p \\
c +\beta^2 a + 2\alpha & \equiv 0 \pmod p \\
\beta^2 b + d - \beta^2 c - \beta^3 a & \equiv 0 \pmod {p^2} \\
-\alpha^2 - \beta^2 ac - bd + 2 \beta a d - \alpha c - \alpha \beta^2 a & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}
Для початку розглянемо випадок, коли $p > 3$. Із другого рівняння знайдемо $\alpha$:
\begin{align*}
\alpha \equiv \beta^2 a - \beta b \pmod p
\end{align*}
Підставивши $\alpha$ в перше рівняння та знизивши четверте рівняння до $\pmod p$, отримуємо два рівняння, які віднімемо -- у результаті
\begin{align*}
c \beta (\beta - 1) \equiv 0 \pmod p
\end{align*}
Отримали необхідну умову для розв'язання системи.
\begin{lemma}
Нехай задано кубічне кільце $\Lambda$ при $p \nmid a,\ p \nmid c$. Припустимо $\Lambda$ допускає розширення елементом $\theta = \dfrac{\alpha + \beta \omega_1 + \omega_2}{p}$. Тоді $\beta = 0$ або $\beta = 1$.
\end{lemma}

Отже, кільце $\Lambda$ при $p \nmid a,\ p \nmid c$, можливо, розширюється елементами вигляду $\theta = \dfrac{\alpha + \omega_2}{p}$ або $\theta = \dfrac{\alpha + \omega_1 + \omega_2}{p}$. Інших варіантів нема.

\subsection*{$\beta = 0$}
Підставивши дане значення, отримуємо систему:
\begin{equation*}
\begin{cases}
d & \equiv 0 \pmod p \\
\alpha & \equiv 0 \pmod p \\
c & \equiv 0 \pmod p \\
d & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}
Оскільки ми розглядали кільце $\Lambda$, де $p \nmid c$, то цей випадок не підходить.

\subsection*{$\beta = 1$}
Підставимо це значення:
\begin{equation*}
\begin{cases}
d - \beta c - \alpha \beta & \equiv 0 \pmod p \\
\alpha & \equiv a - b \pmod p \\
c + a + 2\alpha & \equiv 0 \pmod p \\
 b + d - c - a & \equiv 0 \pmod {p^2} \\
-\alpha^2 - ac - bd + 2 \beta a d - \alpha c - \alpha a & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}
Перше рівняння після підставлення $\alpha$ дає $b+d-c-a \equiv 0 \pmod p$, яке можна опустити через четверте рівняння. Також підставимо $\alpha$ в третє рівняння.
\begin{equation*}
\begin{cases}
\alpha & \equiv a - b \pmod p \\
c + 3a - 2b & \equiv 0 \pmod p \\
 b + d - c - a & \equiv 0 \pmod {p^2} \\
-\alpha^2 - ac - bd + 2 \beta a d - \alpha c - \alpha a & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}
Залишилося підставити $\alpha \equiv a-b + pm \pmod {p^2}$ в останнє рівняння. Враховуючи решта рівнянь системи, отримуємо наступне:
\begin{align*}
b (a-b+c-d) \equiv 0 \pmod {p^2}
\end{align*}
Отже, останнє рівняння виконується автоматично.

\begin{proposition}
Нехай задано кубічне кільце $\Lambda$ при $p \nmid c$.\\
Кільце $\Lambda$ допускає розширення елементом $\theta = \dfrac{\alpha + \omega_1 + \omega_2}{p}$, де число $\alpha \equiv a -b \pmod p \iff \begin{cases} p \mid c + 3a - 2b \\ p^2 \mid -a+b-c+d \end{cases}$.
\end{proposition}

Тепер розглянемо кільце $\Lambda$ при $p \mid c$. Система рівнянь матиме такий вигляд:
\begin{equation*}
\begin{cases}
d - \beta^3 a + \beta^2 b & \equiv 0 \pmod p \\
\alpha & \equiv \beta^2 a - \beta b \pmod p \\
3 \beta^2 a & \equiv 2 \beta b \pmod p \\
\beta^2 b + d - \beta^2 c - \beta^3 a & \equiv 0 \pmod {p^2} \\
-\alpha^2 - \beta^2 a c - bd + 2 \beta a d - \alpha c - \alpha \beta^2 a & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}
Із третього рівняння випливає, що або $\beta = 0$ та $\alpha = 0$, або дана змінна шукається за рівнянням $3 \beta a \equiv 2 b \pmod p$. Для другого випадку, якщо друге рівняння домножимо на $9a$, отримуємо
\begin{align*}
9 a \alpha \equiv (3 \beta a)^2 - (3 \beta a) 3 \beta \pmod p \\
9 a \alpha \equiv -2b^2 \pmod p
\end{align*}

\begin{lemma}
Нехай задано кубічне кільце $\Lambda$ при $p \nmid a,\ p \mid c$. Припустимо $\Lambda$ допускає розширення елементом $\theta = \dfrac{\alpha + \beta \omega_1 + \omega_2}{p}$. Тоді $\theta = \dfrac{\omega_2}{2}$ або $\theta = \dfrac{\alpha + \beta \omega_1 + \omega_2}{2}$ при $\begin{cases} 9 a \alpha \equiv -2b^2 \pmod p \\  3 \beta a \equiv 2 b \pmod p \end{cases}$.
\end{lemma}

\subsubsection*{$\beta = 0$} Тоді система дає обмеження $p \mid d^2$.
\begin{proposition}
Нехай $\Lambda$ -- кубічне кільце при $p \nmid a,\ p \mid c$.\\
Кільце $\Lambda$ допускає розширення елементом $\theta = \dfrac{\omega_2}{p} \iff p^2 \mid d$. 
\end{proposition}

\subsubsection*{$3 a \beta \equiv 2 b \pmod p$} Третє рівняння домножимо на $27a^2$ -- отримуємо
\begin{align*}
27a^2 d - (3 \beta a)^3 + 3b (3 \beta a)^2 \equiv 0 \pmod p \\
27 a^2 d + 4 b^3 \equiv 0 \pmod p
\end{align*}
Перепишемо систему рівнянь:
\begin{equation*}
\begin{cases}
27 a^2 d + 4 b^3 & \equiv 0 \pmod p \\
9 a \alpha & \equiv -2 b^2 \pmod p \\
3 a \beta & \equiv 2 b \pmod p \\
\beta^2 b + d - \beta^2 c - \beta^3 a & \equiv 0 \pmod {p^2} \\
-\alpha^2 - \beta^2 a c - bd + 2 \beta a d - \alpha c - \alpha \beta^2 a & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}
Зробимо підстановку $3a \beta$ та $9a \alpha$ в останні два рівняння. Для цього піднімемо їх:
\begin{align*}
3 a \beta \equiv 2b + pk \pmod {p^2} \\
9a \alpha \equiv -2b^2 + pm \pmod {p^2}
\end{align*}
Четверте рівняння домножимо на $27a^2$ -- після перетворень отримуємо
\begin{align*}
27 a^2 d + 4b^3 - 12 b^2 c \equiv 0 \pmod {p^2}
\end{align*}
Отже, перше рівняння системи можна опустити.

Останнє рівняння домножимо на $81a^2$ -- в результаті:
\begin{align*}
-(9 a \alpha)^2 - 9 a c (3a\beta)^2 - 81 a^2bd + 54 a^2 d (3\beta a) - 9 ac (9 a \alpha) - (9 a \alpha) (3 a \beta)^2 \\ \equiv 0 \pmod {p^2}
\end{align*}
Підставивши та розкривши дужки, одержимо
\begin{align*}
27a^2 bd + 54a^2 dkp - 18ab^2 c + 4b^4 + 8b^3 kp \equiv 0 \pmod {p^2}
\end{align*}
Згрупуємо доданки таким чином:
\begin{align*}
b (27a^2 d + 4b^3) - 18ab^2c + (4b^3 + 27a^2d) 2k p \equiv 0 \pmod {p^2}
\end{align*}
Знаючи, що $27a^2 d + 4b^3 \equiv 12 b^2c \pmod {p^2}$, підставимо це.
\begin{align*}
12b^3 c - 18ab^2c \equiv 0 \pmod {p^2}
\end{align*}
Доданок $12b^2 c \cdot 2kp$ зник, оскільки $pc \equiv 0 \pmod {p^2}$. Отже, система має вигляд:
\begin{equation*}
\begin{cases}
9 a \alpha & \equiv -2 b^2 \pmod p \\
3 a \beta & \equiv 2 b \pmod p \\
27 a^2 d + 4b^3 - 12 b^2 c & \equiv 0 \pmod {p^2} \\
b^2 c (2b - 3a) & \equiv 0 \pmod {p^2}
\end{cases}
\end{equation*}

\begin{proposition}
Нехай $\Lambda$ -- кубічне кільце при $p \nmid a,\ p \mid c$.\\
Кільце $\Lambda$ допускає розширення елементом $\theta = \dfrac{\alpha + \beta \omega_1 + \omega_2}{p}$ при $\begin{cases} 9 a \alpha & \equiv -2 b^2 \pmod p \\
3 a \beta & \equiv 2 b \pmod p \end{cases} \iff \begin{cases} p^2 \mid 27 a^2 d + 4b^3 - 12b^2 c \\ p^2 \mid b^2 c (2b-3a) \end{cases}$.
\end{proposition}

\section{Розв'язання системи конгруентних рівнянь}
\subsection{Випадок, коли $p$ -- просте число, де $p \neq 3$}

Маємо рівняння
\begin{align*}
bt \equiv p-3 \pmod p.
\end{align*}
Дане рівняння має розв'язок $\iff \gcd(b,p) \mid p-3$. Оскільки $p$ -- просте, то $\gcd(b,p) \in \{1,p\}$. Якби $\gcd(b,p) = p$, то отримали би $p \mid p-3$, що неможливо.

Отже, якщо $p \nmid b$, то вищезгадане рівняння має розв'язок. При цьому оскільки $\gcd(b,p) = 1$, то рівняння має лише єдиний розв'язок вигляду
\begin{align*}
t \equiv -3b^{-1} \pmod p
\end{align*}

Єдиний можливий розв'язок необхідно підставити в інші два рівняння системи. Для початку піднімемо розв'язок до $\pmod {p^2}$ -- отримуємо наступне:
\begin{align*}
t \equiv -3b^{-1} + kp \pmod {p^2},
\end{align*}
де $k \in \{0,1,\dots,p-1\}$. Щойно отримане $t$ підставимо в друге рівняння:
\begin{align*}
act^2 + 2bt + 3 & \equiv ac(-3b^{-1} + kp)^2 + 2b(-3b^{-1}+kp) + 3 \\
& \equiv ac(k^2p^2 - 6kpb^{-1} + 9b^{-2}) + 2b(kp - 3b^{-1}) + 3 \\
& \equiv -6kp acb^{-1} + 9 acb^{-2} + 2bkp - 3 \equiv 0 \pmod {p^2}
\end{align*}
Домножимо обидві частини на $b^2$ -- отримуємо:
\begin{align*}
-6kpabc + 9ac + 2b^3kp - 3b^2 & \equiv 2bkp(b^2 - 3ac) - 3(b^2 - 3ac) \\
& \equiv (b^2 - 3ac)(2bkp -1) \equiv 0 \pmod {p^2}
\end{align*}
Зауважимо, що $\gcd(2bkp-1, p^2) = 1$, тому можемо скоротити рівняння:
\begin{align*}
b^2 - 3 ac \equiv 0 \pmod {p^2}
\end{align*}

Отже, якщо $p \nmid b$, а також $b^2 \equiv 3ac \pmod {p^2}$, то наш єдиний розв'язок задовольняє систему із останніх двох рівнянь. Причому $t \equiv -3b^{-1} + kp \pmod {p^2}$ не залежить від параметру $k$.

Зробимо аналогічно для першого рівняння системи, тільки замість $\pmod {p^3}$ розглядається $\pmod {p^2}$ -- маємо:
\begin{multline*}
a^2 dt^3 + act^2 + bt + 1 \\ \equiv a^2 d (-3b^{-1} + kp)^3 + ac (-3b^{-1} + kp)^2 + b (-3b^{-1} + kp) + 1 \\
\equiv a^2 d (-27 b^{-3} + 27kp b^{-2}) + ac (9b^{-2} - 6b^{-1}kp) + b(-3b^{-1} + kp) + 1 \\ \equiv 0 \pmod {p^2}
\end{multline*}
Домножимо обидві частини рівності на $b^{-3}$ -- із урахуванням рівності $b^2 \equiv 3 ac \pmod {p^2}$ отримуємо:
\begin{multline*}
a^2 d(-27 + 27 kpb) + ac(9b - 6b^2 kp) + b^4(-3b^{-1} + kp) + b^3 \\
\equiv -27a^2d  + 27 a^2 b d k p + 9abc - 6 ab^2 c kp - 3b^3 + b^4 kp + b^3 \\
\equiv -27a^2d + 27 a^2 b d k p + 3b^3 - 2 b^4 kp - 3b^3 + b^4 kp + b^3 \\
\equiv -27a^2d (1 - bkp) + b^3(1 - bkp) \equiv (1-bkp)(b^3 - 27a^2d) \equiv 0 \pmod {p^2}
\end{multline*}
Зауважимо, що $\gcd(1-bkp, p^2) = 1$, тому можемо скоротити рівняння:
\begin{align*}
b^3 - 27a^2 d \equiv 0 \pmod {p^2}
\end{align*}

Нарешті, розглянемо перше рівняння:
\begin{align*}
a^2 dt^3 + act^2 + bt + 1 \equiv 0 \pmod {p^3}
\end{align*}
Оскільки $t \equiv -3b^{-1} + kp \pmod {p^2}$ буде розв'язком системи останніх двох рівнянь при вищезгаданих умов, причому нема залежності від $k$, то для спрощення оберемо $t \equiv -3b^{-1} \pmod {p^2}$. Піднімемо це рівняння до $\pmod {p^3}$ -- отримуємо наступне:
\begin{align*}
t \equiv -3b^{-1} + lp^2 \pmod {p^3},
\end{align*} 
де $l \in \{0,1,\dots,p-1\}$. Щойно отримане $t$ підставимо в перше рівняння:
\begin{multline*}
a^2 dt^3 + act^2 + bt + 1 \\ 
\equiv a^2 d (-3b^{-1} + lp^2)^3 + ac(-3b^{-1} + lp^2)^2 + b(-3b^{-1} + lp^2) + 1 \\
\equiv a^2d (-27b^{-3} + 27b^{-2} lp^2) + ac (9b^{-2} - 6 l b^{-1} p^2) + b (-3b^{-1} + lp^2) + 1 \\ 
\equiv 0 \pmod {p^3}
\end{multline*}
Домножимо обидві частини рівності на $b^3$ -- отримуємо:
\begin{multline*}
a^2d (-27 + 27b lp^2) + ac (9b - 6 l b^2 p^2) + (-3 b^3 + lp^2 b^4) + b^3 \\ 
\equiv 0 \pmod {p^3}
\end{multline*}
Пригадаємо, що необхідно $b^2 \equiv ac \pmod {p^2}$, а також $27a^2 d \equiv b^3 \pmod {p^2}$. Підмінаючи до $\pmod {p^3}$, отримуємо:
\begin{align*}
3ac \equiv b^2 + \xi p^2 \pmod {p^3} \\
27a^2 d \equiv b^3 + \eta p^2 \pmod {p^3},
\end{align*}
де $\xi,\eta \in \{0,1,\dots,p-1\}$. Внаслідок чого
\begin{multline*}
27 a^2d (-1+ b lp^2) + 3ac (3b - 2 l b^2 p^2) + (-3 b^3 + lp^2 b^4) + b^3 \\ \equiv (b^3 + \eta p^2)(-1 + b lp^2) + (b^2 + \xi p^2)(3b - 2 l b^2 p^2) + (-3 b^3 + lp^2 b^4) + b^3 \\ \equiv -b^3 + b^4 lp^2 - \eta p^2 + 3b^3 - 2b^4 lp^2 + 3 \xi p^2 b - 3b^3 + b^4 lp^2 + b^3 \\
\equiv - \eta p^2 + 3\xi p^2 b \equiv 0 \pmod {p^3}
\end{multline*}
Отже, ми отримали наступне:
\begin{align*}
\eta p^2 \equiv 3 \xi p^2 b \pmod {p^3}
\end{align*}
Тоді зауважимо, що
\begin{align*}
27 a^2 d \equiv b^3 + \eta p^2 \equiv b^3 + 3 \xi p^2 b \equiv b^3 + 3b(3ac - b^2) \equiv -2b^3 + 9abc \pmod {p^3}
\end{align*}
Підсумуємо:

\begin{lemma}
\label{system_when_p_is_prime_and_bigger_than_three}
Нехай $p > 3$ -- просте число або $p = 2$. Припустимо, що $p \nmid b,\ b^2 \equiv 3ac \pmod {p^2}$, а також $27a^2d \equiv 9abc - 2b^3 \pmod {p^3}$. Тоді система конгруентних рівнянь \eqref{system_congruent_equations} має єдиний розв'язок $t \equiv -3 b^{-1} \pmod {p}$, де число $b^{-1}$ -- оборотний елемент класу лишків $\mathbb{Z}_p$ відносно множення.
\end{lemma}

Пригадаємо, що кубічному кільцю відповідає кубічна форма, дискримінант якої обчислюється за формулою
\begin{equation}
\Delta = b^2 c^2 - 4c^3 a - 4 b^3 d + 18 abcd - 27a^2d^2
\end{equation}
Сприймаючи дискримінант за функцію від чотирьох змінних, знайдемо частинну похідну за аргументом $d$:
\begin{align*}
\Delta'_d = -4b^3 + 18 abc - 54a^2 d
\end{align*}
Тоді умову $27a^2d \equiv 9abc - 2b^3 \pmod {p^3}$ після множення на $2$ можна записати таким чином:
\begin{align*}
\Delta'_d \equiv 0 \pmod {p^3}
\end{align*}

\subsection{Випадок, коли $p = 3$}. Третє рівняння системи спроститься до вигляду
\begin{align*}
bt \equiv 0 \pmod 3
\end{align*}

Припустимо, що $3 \nmid b$. Тоді система може має єдиний розв'язок $t \equiv 0 \pmod 3$. Якщо піднятися до $\pmod 9$, то в нас існують три опції:
\begin{align*}
t \equiv 0 \pmod 9,\qquad t \equiv 3 \pmod 9,\qquad t \equiv 6 \pmod 9
\end{align*}
Якщо $t \equiv 0 \pmod 9$, то із другого рівняння випливає, що $3 \equiv 0 \pmod 9$, що неможливо. Якщо $t \equiv 3 \pmod 9$, то із першого рівняння випливає, що
\begin{align*}
3b \equiv 8 \pmod 9
\end{align*}
Оскільки $\gcd(3,9) = 3 \nmid 8$, то розв'язків не має -- неможливо. Нарешті, якщо $t \equiv 6 \pmod 9$, то із першого рівняння випливає, що
\begin{align*}
6b \equiv 8 \pmod 9
\end{align*}
Аналогічними міркуваннями розв'язків нема -- неможливо.

\begin{lemma}
Система конгруентних рівнянь \eqref{system_congruent_equations} не має розв'язків при $3 \nmid b$.
\end{lemma}

Надалі розглядається випадок $3 \mid b$. Третє рівняння автоматично виконується, тому залишається система
\begin{equation}
\label{system_congruent_equations}
    \begin{cases}
      a^2 d t^3 + ac t^2 + bt + 1 & \equiv 0 \pmod {27} \\
      act^2 + 2bt + 3 & \equiv 0 \pmod {9} 
    \end{cases}\,.
\end{equation}
Тимчасово опустимося до системи рівнянь за $\pmod 3$. Зважаючи на умову $3 \mid b$, маємо
\begin{equation*}
\label{system_congruent_equations}
    \begin{cases}
      a^2 d t^3 + ac t^2 + 1 & \equiv 0 \pmod {3} \\
      act^2 & \equiv 0 \pmod {3} 
    \end{cases}\,.
\end{equation*}

Припустимо, що $3 \nmid ac$. Тоді із другого рівняння отримуємо $t \equiv 0 \pmod 3$. Але, підставляючи в перше рівняння, матимемо $1 \equiv 0 \pmod 3$, що неможливо.

Отже, надалі $3 \mid ac$. Звідси ми маємо $3 \mid a$ або $3 \mid c$.

Якщо $3 \mid a$, то знову $1 \equiv 0 \pmod 3$ із першого рівняння, що неможливо.

\begin{lemma}
Система конгруентних рівнянь \eqref{system_congruent_equations} не має розв'язків при $3 \nmid ac$ або $3 \mid a$.
\end{lemma}

Отже, залишається випадок, коли $3 \mid c$. Значить, перше рівняння дає
\begin{align*}
a^2 dt^3 \equiv 2 \pmod 3
\end{align*}
Зауважимо, що розв'язок існує при $3 \nmid a^2d$.
\begin{corollary}
Система конгруентних рівнянь \eqref{system_congruent_equations} не має розв'язків при $3 \mid d$.
\end{corollary}

У нас існують два варіанти за $\pmod 3$: $a^2 d \in \{1,2\}$.

Нехай $a^2 d \equiv 1 \pmod 3$, тоді маємо $t^3 \equiv 2 \pmod 3$, внаслідок чого отримуємо $t \equiv 2 \pmod 3$ -- єдиний кандидат на розв'язок системи \eqref{system_congruent_equations} з урахуванням певних умов.

Нехай $a^2 d \equiv 2 \pmod 3$, тоді маємо $2t^3 \equiv 2 \pmod 3$, внаслідок чого отримуємо $t \equiv 1 \pmod 3$ -- також єдиний кандидат.

Умову $a^2 d \equiv 1 \pmod 3$ можна трошки детально розписати. Уже відомо, що $3 \nmid d$, а тому звідси $d \in \{1,2\}$, якщо брати за $\pmod 3$. Якщо $d = 2$, то звідси $2a^2 \equiv 1 \pmod 3$, або $a^2 \equiv 2 \pmod 3$. Проте зазначимо, що $2$ не є квадратичним лишком $\pmod 3$, тому розв'язків відносно $a$ не має. Водночас якщо $d = 1$, то $a^2 \equiv 1 \pmod 3$, де існує розв'язок.

Аналогічними міркуваннями розглянемо $a^2 d \equiv 2 \pmod 3$. Цього разу при $d = 1$ не знайдеться розв'язку відносно $a$, коли водночас при $d = 2$ розв'язок існує.

Підіб'ємо певний підсумок: 

\begin{proposition}
Нехай $3 \nmid a,\ 3 \mid b,\  3 \mid c$. Тоді із системи рівнянь \eqref{system_congruent_equations} випливають можливі розв'язки: 
\begin{enumerate}
\item $t \equiv 2 \pmod 3$ за умовою, що $d \equiv 1 \pmod 3$;
\item $t \equiv 1 \pmod 3$ за умовою, що $d \equiv 2 \pmod 3$.
\end{enumerate}
\end{proposition}

\subsubsection*{Експериментальні результати}
Піднімаючись до $\!\! \pmod 27$, ми маємо вісімнадцять потенційних розв'язків у залежності від параметра $d$: \begin{align*}
t \in \{1,4,7,10,13,16,19,22,25\} \text{ при } d \equiv 2 \pmod 3 \\
t \in \{2,5,8,11,14,17,20,23,26\} \text{ при } d \equiv 1 \pmod 3
\end{align*}

\subsubsection*{Теоретичний пошук}
Маючи накладені умови з твердження, перевіримо, чи будуть вищезгадані $t$ дійсно розв'язками системи.

\subsubsection{Випадок $t \equiv 2 \pmod 3$}
Отже, в цьому випадку $a^2d \equiv 1 \pmod 3$.

Спочатку піднімемося до $\pmod 9$. Тоді маємо три потенційних кандидати: $t \in \{2,5,8\}$.

Нехай $t \equiv 2 \pmod 9$. Маємо наступне:
\begin{equation*}
    \begin{cases}
      8 a^2 d  + 4 ac + 2b + 1 & \equiv 0 \pmod {27} \\
      4ac + 4b + 3 & \equiv 0 \pmod {9} 
    \end{cases}\,.
\end{equation*}
Оскільки $a^2 d \equiv 1 \pmod 3$, то звідси за $\pmod 9$ маємо $a^2 d \in \{1,4,7\}$. Якщо систему рівнянь спустити до $\pmod 9$, отримуємо три варіанти
\begin{align*}
    \begin{cases}
      8  + 4 ac + 2b + 1 & \equiv 0 \pmod {9} \\
      4ac + 4b + 3 & \equiv 0 \pmod {9} 
    \end{cases}\, \\    \begin{cases}
      32  + 4 ac + 2b + 1 & \equiv 0 \pmod {9} \\
      4ac + 4b + 3 & \equiv 0 \pmod {9} 
    \end{cases}\, \\     \begin{cases}
      49  + 4 ac + 2b + 1 & \equiv 0 \pmod {9} \\
      4ac + 4b + 3 & \equiv 0 \pmod {9} 
    \end{cases}\,.
\end{align*}
Розглянемо першу систему. Ми отримуємо $4ac \equiv -2b \pmod 9$. Значить, звідси
\begin{align*}
4ac + 4b + 3 \equiv 2b + 3 \equiv 0 \pmod 9
\end{align*}
Останнє можливе лише при $b \equiv 3 \pmod 9$. Також маємо $4ac \equiv -6 \equiv 3 \pmod 9$.

\section{Про зображення кубічних кілець}
Нехай $\Lambda$ -- кубічне кільце та $M \subset \Lambda$ -- максимальне надкільце. Можемо записати факторкільце
\begin{align*}
M/_\Lambda = \mathbb{Z}_{n_1(\Lambda)} \oplus \mathbb{Z}_{n_2(\Lambda)},
\end{align*}
де числа $n_1(\Lambda) \mid n_2(\Lambda)$.

\begin{theorem}
Кубічне кільце $\Lambda$ має скінченне число неізоморфних нерозкладених зображень $\iff n_1(\Lambda)$ -- вільне від квадратів.
\end{theorem}

\section{Частинний випадок максимального кубічного кільця}
Нехай $\Lambda$ -- кубічне кільце, яке задається числами $(a,b,c,d)$. Продемонструємо кілька можливих частинних випадків.

Розширимо кільце $\Lambda$ так, щоб $\{1,\omega_1,\theta\}$ був базисом кільця $\tilde{\Lambda} \supset \Lambda$, де елемент $\theta = \dfrac{\alpha + \beta \omega_1 + \omega_2}{p}$. Тобто маємо $\theta = \dfrac{1 + \omega_1 + \omega_2}{p}$. Із точністю до $\pmod {2^2}$ нам треба обрати такі значення $a,b,c,d$ кільця $\Lambda$: 
\begin{center}
\begin{tabular}{c|c|c|c}
$a$ & $b$ & $c$ & $d$ \\
$1$ & $0$ & $1$ & $2$ \\
$1$ & $0$ & $3$ & $0$ \\
$1$ & $2$ & $3$ & $2$
\end{tabular}
\end{center}
Далі необхідно нормалізувати базис $\{1,\omega_1,\theta\}$. Утім уже відомо яким чином. Лише розпишемо, чому дорівнюють оновлені значення $\tilde{a},\tilde{b},\tilde{c},\tilde{d}$.
\begin{center}
\begin{tabular}{c|c|c|c}
$\tilde{a}$ & $\tilde{b}$ & $\tilde{c}$ & $\tilde{d}$ \\
$2$ & $-3$ & $2$ & $0$ \\
$2$ & $-3$ & $3$ & $-1$ \\
$2$ & $-1$ & $1$ & $0$
\end{tabular}
\end{center}
Кільце $\tilde{\Lambda}_3$, (що визначається числами третього рядка) більше не матиме розширення. Кільце $\tilde{\Lambda}_2$, можливо, має розширення при $p = 3$. Кільце $\tilde{\Lambda}_1$, можливо, ще раз має розширення при $p = 2$. Це можна пояснити отриманими значеннями дискримінанту $\Delta_{\tilde{\Lambda}}$: $28, -27, -7$.

Розглянемо $\tilde{\Lambda}_1 = \left<1,\omega_1,\omega_2\right>$. Для зручності опускатимемо тільда. Маємо $(a,b,c,d) = (2,-3,2,0)$. Ми можемо розширитися до кільце $\tilde{\tilde{\Lambda}}$ за базисом $\{1,\omega_1, \theta\}$, де елемент $\theta = \dfrac{\omega_2}{2}$ (тут $\alpha = \beta = 0$). Отже, матимемо кільце $\tilde{\tilde{\Lambda}} = \left< 1,\omega_1, \dfrac{\omega_2}{p} \right>$. Тепер кільце задається числами $(4,-3,1,0)$. Оскільки далі $\Delta_{\tilde{\tilde{\Lambda}}} = -7$, то більше розширень нема.

Розглянемо кільце $\tilde{\Lambda}_2 = \left<1,\omega_1,\omega_2\right>$. Маємо $(a,b,c,d) = (2,-3,3,-1)$. Ми можемо розширитися до кільця $\tilde{\tilde{\Lambda}}$ за базисом $\{1,\omega_1,\theta\}$, де елемент $\theta = \dfrac{2 + \omega_1 + \omega_2}{3}$ (тут $\alpha = 2, \beta = 1$).

Тут просто запишу всі максимальні кільця.\\
$(1,0,1,2) \leftrightarrow \Lambda \subset M = \left< 1,\omega_1-1,\dfrac{1+\omega_1+\omega_2}{4} \right> \leftrightarrow (4,-2,1,0)$\\
$(2,-3,3,-1) \leftrightarrow \Lambda \subset M = \left< 1, \omega_1-1, \dfrac{3 + 3 \omega_1 + \omega_2}{6} \right> \leftrightarrow (6, -9, 5, -1)$\\
$(1,2,3,2) \leftrightarrow \Lambda \subset M = \left< 1, \omega_1 - 1, \dfrac{-1+\omega_1+\omega_2}{2} \right> \leftrightarrow (2,-1,1,0)$.

Відомо, що $|M/_\Lambda|^2 = \dfrac{\Delta_\Lambda}{\Delta_M}$, звідси маємо порядок відповідно $4,\ 6,\ 2$. Оскільки $M/_\Lambda$ є абелевою групою, то ми можемо подати її в пряму суму циклічних.

У першому випадку $|M/_\Lambda| = 4$, утім звідси $M/_\Lambda$ буде ізоморфною або $\mathbb{Z}_4$, або $\mathbb{Z}_2 \times \mathbb{Z}_2$. Пригадаємо, що якщо маємо вкладення $\mathbb{Z}^3 \hookrightarrow \mathbb{Z}^3$, то їхні базисні елементи мають бути пропорційними. Зокрема базис $\left\{ 1,\ \omega_1-1,\ \dfrac{1+\omega_1+\omega_2}{4} \right\}$ кільця $M$ пропорційний базису $\{1, \omega_1-1, 1+\omega_1+\omega_2\}$ кільця $\Lambda$. Відображення $\Lambda \hookrightarrow M$ можна задати ось таким чином: $(x,y,z) \mapsto (x,y,4z)$. Звідси $\Lambda \cong \mathbb{Z} \oplus \mathbb{Z} \oplus 4 \mathbb{Z}$, а також $M \cong \mathbb{Z} \oplus \mathbb{Z} \oplus \mathbb{Z}$. Фактор $M/_\Lambda$ буде прямою сумою факторів кожного додану, отримуємо $M/_\Lambda \cong \mathbb{Z}/_{4 \mathbb{Z}} = \mathbb{Z}_4 \cong \mathbb{Z}_1 \oplus \mathbb{Z}_4$. Зауважимо, що $1 \mid 4$, а також $1$ -- вільне від квадратів число. Отже, кубічне кільце $\Lambda$  має скінченне число неізоморфних та нерозкладених зображень.

Аналогічно для другого випадку $M/_\Lambda \cong \mathbb{Z}_1 \oplus \mathbb{Z}_6$ та $M/_\Lambda \cong \mathbb{Z}_1 \oplus \mathbb{Z}_2$. У кожному з цих випадків кубічне кільце $\Lambda$ задовольняє теорему.

\begin{proposition}
Нехай кубічне кільце $\Lambda$ має розширення $\tilde{\Lambda}$, причому $[\tilde{\Lambda}: \Lambda] = p$. Тоді $\Delta_\Lambda = p^2 \Delta_{\tilde{\Lambda}}$. Іншими словами, $p^2 \mid \Delta_\Lambda$.

На жаль, у зворотний бік дане твердження не працює. Зокрема можна розглянути кубічне кільце $\Lambda$, яке задається числами $(0,6,5,0)$. Його дискримінант $\Delta_\Lambda = 900$, причому $2^2 \mid 900$. Утім не існує розширення $\tilde{\Lambda}$, щоб було $[\tilde{\Lambda}:\Lambda] = 2$.
